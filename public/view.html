🛠️ Como Resolver:

Recomendação: Altere suas rotas no Express para usar o plural (Opção 1)

// No seu arquivo index.js, altere de:
app.use('/api/aluno', AlunoR);
app.use('/api/plano', PlanoR);
app.use('/api/avaliacao', AvaliacaoR);
app.use('/api/matricula', MatriculaR);

// Para:
app.use('/api/alunos', AlunoR);
app.use('/api/planos', PlanoR);
app.use('/api/avaliacoes', AvaliacaoR);
app.use('/api/matriculas', MatriculaR);

Alternativa: Altere as URLs no frontend para usar o singular

// No seu JavaScript, altere de:
const ENDPOINTS = {
  alunos: '/api/alunos',
  planos: '/api/planos',
  avaliacoes: '/api/avaliacoes',
  matriculas: '/api/matriculas'
};

const ENDPOINTS = {
  alunos: '/api/aluno',
